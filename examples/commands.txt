# --- Create a database (password piped in, twice) --------------

# With a key file as well (generates it)
$ keepassxc-cli db-create --set-password \
    --set-key-file vault-key.key keyvault.kdbx

# --- Groups and entries ------------------------------------------
$ keepassxc-cli mkdir vault.kdbx Email
$ keepassxc-cli add vault.kdbx "Email/Example Mail" \
    -u reader@example.com -p --url https://mail.example.com
$ keepassxc-cli add vault.kdbx "Finance/Example Bank" \
    -u saver@example.com -g -L 20 --url https://bank.example.com
$ keepassxc-cli ls -R vault.kdbx
$ keepassxc-cli show vault.kdbx "Email/Example Mail"
Password: <redacted>
$ keepassxc-cli search vault.kdbx example
$ keepassxc-cli edit --url https://router.example.com vault.kdbx "Web/Example Router"
$ keepassxc-cli mv vault.kdbx "Web/Example Router" Finance
$ keepassxc-cli rm vault.kdbx "Web/Throwaway"
Successfully recycled entry Throwaway.   # it is in the Recycle Bin now
