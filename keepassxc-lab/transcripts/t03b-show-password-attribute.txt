# The summary view of `show` redacts the password even with -s (t03).
# Requesting the Password attribute explicitly, with -s, reveals it:

$ echo "$LAB_PW" | keepassxc-cli show -s -a Password vault.kdbx "Email/Example Mail"
Enter password to unlock vault.kdbx: 
S3cure!Demo

$ echo "$LAB_PW" | keepassxc-cli show -s -a Password vault.kdbx "Finance/Example Bank"   (the generated one)
Enter password to unlock vault.kdbx: 
ikjzz24NrKFzUYVAVYMy

$ echo "$LAB_PW" | keepassxc-cli show --all -s vault.kdbx "Email/Example Mail"
